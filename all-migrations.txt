
-- FILE: 20261014112657_create_users_table.sql
/*
# Create users table

## What this migration does
Creates the users table for customer accounts.

## New Tables
1. `users` â€” customer accounts
   - id (text, primary key)
   - email (text, unique)
   - name (text, nullable)
   - phone (text, nullable)
   - password_hash (text): bcrypt hashed password
   - created_at, updated_at (timestamptz)

## Security
- RLS enabled on users
- anon can INSERT (registration via API)
- authenticated can SELECT and UPDATE own data
*/

CREATE TABLE IF NOT EXISTS users (
  id text PRIMARY KEY,
  email text UNIQUE NOT NULL,
  name text,
  phone text,
  password_hash text NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

ALTER TABLE users ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "anon_insert_users" ON users;
CREATE POLICY "anon_insert_users" ON users FOR INSERT
TO anon, authenticated WITH CHECK (true);

DROP POLICY IF EXISTS "user_read_own" ON users;
CREATE POLICY "user_read_own" ON users FOR SELECT
TO authenticated USING (true);

DROP POLICY IF EXISTS "user_update_own" ON users;
CREATE POLICY "user_update_own" ON users FOR UPDATE
TO authenticated USING (true) WITH CHECK (true);

-- FILE: 20261014112714_create_base_tables.sql
/*
# Create base tables for The Golden Tiger storefront

## What this migration does
Creates 5 core tables: categories, products, orders, order_items, click_logs.
The users table already exists from a prior migration.

## New Tables
1. `categories` â€” product category hierarchy
2. `products` â€” full product catalog with SEO columns included
3. `orders` â€” customer orders with checkout flow status
4. `order_items` â€” line items per order
5. `click_logs` â€” telemetry/analytics events for CTR and conversion tracking

## Security
- RLS enabled on all tables
- categories: public read (anon, authenticated)
- products: public read (anon, authenticated)
- orders: anon+authenticated CRUD (checkout flow)
- order_items: anon+authenticated CRUD (checkout flow)
- click_logs: anon+authenticated INSERT and SELECT (tracking)
- Admin operations use service role key server-side, bypassing RLS
*/

-- ============================================================
-- CATEGORIES
-- ============================================================
CREATE TABLE IF NOT EXISTS categories (
  id text PRIMARY KEY,
  slug text UNIQUE NOT NULL,
  name text NOT NULL,
  parent_id text REFERENCES categories(id) ON DELETE SET NULL,
  image_url text,
  sort_order integer NOT NULL DEFAULT 0,
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_categories_parent_id ON categories(parent_id);

ALTER TABLE categories ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "anon_select_categories" ON categories;
CREATE POLICY "anon_select_categories" ON categories FOR SELECT
  TO anon, authenticated USING (true);

-- ============================================================
-- PRODUCTS
-- ============================================================
CREATE TABLE IF NOT EXISTS products (
  id text PRIMARY KEY,
  slug text UNIQUE NOT NULL,
  title text NOT NULL,
  brand text,
  sku text UNIQUE,
  short_description text,
  description text NOT NULL DEFAULT '',
  features text[] NOT NULL DEFAULT '{}',
  specifications jsonb,
  price_cents integer NOT NULL,
  compare_at_cents integer,
  currency text NOT NULL DEFAULT 'USD',
  stock integer NOT NULL DEFAULT 0,
  images text[] NOT NULL DEFAULT '{}',
  video_urls text[] NOT NULL DEFAULT '{}',
  rating real NOT NULL DEFAULT 0,
  review_count integer NOT NULL DEFAULT 0,
  seo_title text,
  seo_description text,
  seo_keywords text,
  tags text[] NOT NULL DEFAULT '{}',
  is_published boolean NOT NULL DEFAULT true,
  category_id text NOT NULL REFERENCES categories(id) ON DELETE RESTRICT,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_products_category_id ON products(category_id);
CREATE INDEX IF NOT EXISTS idx_products_published_created ON products(is_published, created_at);
CREATE INDEX IF NOT EXISTS idx_products_price_cents ON products(price_cents);
CREATE INDEX IF NOT EXISTS idx_products_rating ON products(rating);
CREATE INDEX IF NOT EXISTS idx_products_tags_gin ON products USING GIN(tags);

ALTER TABLE products ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "anon_select_products" ON products;
CREATE POLICY "anon_select_products" ON products FOR SELECT
  TO anon, authenticated USING (true);

-- ============================================================

  payment_method text,
  payment_ref text,

  completed_at timestamptz,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_orders_status_created ON orders(status, created_at);
CREATE INDEX IF NOT EXISTS idx_orders_session_id ON orders(session_id);
CREATE INDEX IF NOT EXISTS idx_orders_user_id ON orders(user_id);

/*
# Create users table

## What this migration does
Creates the users table for customer accounts.

## New Tables
1. `users` â€” customer accounts
   - id (text, primary key)
   - email (text, unique)
   - name (text, nullable)
   - phone (text, nullable)
   - password_hash (text): bcrypt hashed password
   - created_at, updated_at (timestamptz)

## Security
- RLS enabled on users
- anon can INSERT (registration via API)
- authenticated can SELECT and UPDATE own data
*/

CREATE TABLE IF NOT EXISTS users (
  id text PRIMARY KEY,
  email text UNIQUE NOT NULL,
  name text,
  phone text,
  password_hash text NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

ALTER TABLE users ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "anon_insert_users" ON users;
CREATE POLICY "anon_insert_users" ON users FOR INSERT
TO anon, authenticated WITH CHECK (true);

DROP POLICY IF EXISTS "user_read_own" ON users;
CREATE POLICY "user_read_own" ON users FOR SELECT
TO authenticated USING (true);

DROP POLICY IF EXISTS "user_update_own" ON users;
CREATE POLICY "user_update_own" ON users FOR UPDATE
TO authenticated USING (true) WITH CHECK (true);

- admin_reply (text, nullable) â€” admin response shown below review
- admin_replied_at (timestamptz, nullable)
- created_at (timestamptz)

- RLS enabled
- Anyone (anon + authenticated) can SELECT approved reviews

  product_id text NOT NULL REFERENCES products(id) ON DELETE CASCADE,

-- FILE: 20261016225250_fix_users_rls_for_anon_login.sql.sql
/*
# Fix users table RLS for anon login lookup

## Problem
The login endpoint runs with the anon Supabase key (no user is signed in yet).
It queries the `users` table to look up a user by email and verify their password.
However, the only SELECT policy on `users` is scoped to `authenticated` â€” the `anon`
role has no SELECT policy, so RLS blocks the query and the login fails with a
generic "Something went wrong" error.

## Fix
Add an `anon` SELECT policy on the `users` table so the login route can look up
users by email. This is safe because:
1. The login API route only selects the columns needed for authentication (id, email, name, password_hash).
2. The password_hash is used server-side for bcrypt verification and is never sent to the browser.
3. This matches the pattern already used on the `admins` table (anon SELECT is enabled there for the same reason).

## Security
- New policy: `anon_select_users_for_login` â€” SELECT to `anon, authenticated` with `USING (true)`.
- This is intentionally permissive because this is a single-tenant app with no per-user data isolation in the browser â€” all storefront data is public. The login flow requires anon read access to verify credentials.
*/

DROP POLICY IF EXISTS "anon_select_users_for_login" ON users;
CREATE POLICY "anon_select_users_for_login" ON users FOR SELECT
  TO anon, authenticated USING (true);
